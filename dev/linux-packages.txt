# Ubuntu packages to build and test the Linux desktop app. Installed by dev/Containerfile and by CI
# (.github/workflows/ci.yml), so both have the same set. One or more per line; `#` starts a comment.
# Build: toolchain, GTK, SQLite, libsecret (flutter_secure_storage links it on Linux)
clang cmake ninja-build pkg-config libgtk-3-dev liblzma-dev libstdc++-12-dev
libsqlite3-dev sqlite3 libsecret-1-dev
# Headless run (tools/build.sh desktop-test / desktop-smoke): X server, software OpenGL (Mesa llvmpipe), session bus
xvfb xauth dbus xdg-user-dirs libgl1-mesa-dri libegl1 libegl-mesa0 libglx-mesa0 libgles2
# Fonts: GTK and Flutter's text fall back to fontconfig's default sans
fontconfig fonts-dejavu-core
# Live mode (tools/live.sh): file pickers. file_picker talks to the xdg-desktop-portal FileChooser on the session bus;
# live mode runs its own private bus with these two on it, so the dialog shows in the app's Wayland window
xdg-desktop-portal xdg-desktop-portal-gtk
